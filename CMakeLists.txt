cmake_minimum_required(VERSION 3.24)
project(Torrent VERSION 0.1.6 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "Torrent requires Windows 11 and the Windows SDK")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(LibtorrentRasterbar CONFIG REQUIRED)
add_executable(Torrent WIN32 src/main.cpp src/engine.cpp assets/app.rc)
target_compile_definitions(Torrent PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_include_directories(Torrent PRIVATE assets)
target_link_libraries(Torrent PRIVATE LibtorrentRasterbar::torrent-rasterbar d2d1 dwrite comctl32 comdlg32 shell32 ole32 uuid dwmapi windowscodecs)
if(MSVC)
  target_compile_options(Torrent PRIVATE /W4 /utf-8 /permissive-)
  target_link_options(Torrent PRIVATE /MANIFEST:NO)
endif()
install(TARGETS Torrent RUNTIME DESTINATION .)
install(FILES assets/fonts/OFL.txt DESTINATION licenses RENAME Montserrat-OFL.txt)
include(CTest)
if(BUILD_TESTING)
  add_executable(engine_test tests/engine_test.cpp src/engine.cpp)
  target_compile_definitions(engine_test PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  target_include_directories(engine_test PRIVATE src)
  target_link_libraries(engine_test PRIVATE LibtorrentRasterbar::torrent-rasterbar)
  if(MSVC)
    target_compile_options(engine_test PRIVATE /W4 /utf-8 /permissive-)
  endif()
  add_test(NAME engine_state COMMAND engine_test)
  set_tests_properties(engine_state PROPERTIES TIMEOUT 120)
endif()

# The setup shell draws the reference design; NSIS remains the installation backend.
if(EXISTS "${CMAKE_SOURCE_DIR}/dist/Torrent-Payload.exe")
  file(TO_CMAKE_PATH "${CMAKE_SOURCE_DIR}" TORRENT_SOURCE)
  configure_file(installer/setup.rc.in ${CMAKE_BINARY_DIR}/setup.rc @ONLY)
  add_executable(TorrentSetup WIN32 src/setup.cpp ${CMAKE_BINARY_DIR}/setup.rc)
  set_target_properties(TorrentSetup PROPERTIES OUTPUT_NAME "Torrent-Setup")
  target_compile_definitions(TorrentSetup PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(TorrentSetup PRIVATE d2d1 dwrite dwmapi windowscodecs ole32 shell32 uuid advapi32)
  if(MSVC)
    target_compile_options(TorrentSetup PRIVATE /W4 /utf-8 /permissive-)
    target_link_options(TorrentSetup PRIVATE /MANIFEST:NO)
  endif()
endif()
