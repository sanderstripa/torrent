cmake_minimum_required(VERSION 3.24)
project(Torrent VERSION 1.4.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "Torrent requires Windows 11 and the Windows SDK")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(LibtorrentRasterbar CONFIG REQUIRED)
add_executable(Torrent WIN32 src/main.cpp src/engine.cpp assets/app.rc)
target_compile_definitions(Torrent PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_include_directories(Torrent PRIVATE assets)
target_link_libraries(Torrent PRIVATE LibtorrentRasterbar::torrent-rasterbar d2d1 dwrite comctl32 comdlg32 shell32 ole32 uuid dwmapi windowscodecs)
option(TORRENT_RUST_EDITION "Enable the experimental Rust size sorter" OFF)
if(TORRENT_RUST_EDITION)
  find_program(CARGO_EXECUTABLE cargo REQUIRED)
  set(RUST_MANIFEST "${CMAKE_CURRENT_SOURCE_DIR}/rust-edition/Cargo.toml")
  set(RUST_TARGET_DIR "${CMAKE_CURRENT_BINARY_DIR}/rust")
  set(RUST_LIB "${RUST_TARGET_DIR}/x86_64-pc-windows-msvc/release/torrent_re_sort.lib")
  add_custom_command(OUTPUT "${RUST_LIB}"
    COMMAND "${CMAKE_COMMAND}" -E env "RUSTFLAGS=-C target-feature=+crt-static"
      "${CARGO_EXECUTABLE}" build --locked --release --manifest-path "${RUST_MANIFEST}"
      --target-dir "${RUST_TARGET_DIR}" --target x86_64-pc-windows-msvc
    DEPENDS "${RUST_MANIFEST}" rust-edition/Cargo.lock rust-edition/src/lib.rs
    VERBATIM)
  add_custom_target(torrent_re_sort DEPENDS "${RUST_LIB}")
  function(torrent_enable_rust target)
    add_dependencies(${target} torrent_re_sort)
    target_compile_definitions(${target} PRIVATE TORRENT_RUST_EDITION)
    target_link_libraries(${target} PRIVATE "${RUST_LIB}" ntdll userenv ws2_32 dbghelp synchronization)
  endfunction()
  torrent_enable_rust(Torrent)
endif()
if(MSVC)
  target_compile_options(Torrent PRIVATE /W4 /utf-8 /permissive-)
  target_link_options(Torrent PRIVATE /MANIFEST:NO)
endif()
install(TARGETS Torrent RUNTIME DESTINATION .)
install(FILES assets/fonts/OFL.txt DESTINATION licenses RENAME Montserrat-OFL.txt)
install(FILES assets/connect-theme/LICENSE DESTINATION licenses RENAME Lucide-ISC.txt)
include(CTest)
if(BUILD_TESTING)
  add_executable(engine_test tests/engine_test.cpp src/engine.cpp)
  target_compile_definitions(engine_test PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  target_include_directories(engine_test PRIVATE src)
  target_link_libraries(engine_test PRIVATE LibtorrentRasterbar::torrent-rasterbar)
  if(MSVC)
    target_compile_options(engine_test PRIVATE /W4 /utf-8 /permissive-)
  endif()
  add_test(NAME engine_state COMMAND engine_test)
  set_tests_properties(engine_state PROPERTIES TIMEOUT 120)
  add_executable(file_list_sort_test tests/file_list_sort_test.cpp)
  target_include_directories(file_list_sort_test PRIVATE src)
  target_compile_definitions(file_list_sort_test PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  if(MSVC)
    target_compile_options(file_list_sort_test PRIVATE /W4 /utf-8 /permissive-)
  endif()
  if(TORRENT_RUST_EDITION)
    torrent_enable_rust(file_list_sort_test)
  endif()
  add_test(NAME file_list_sort COMMAND file_list_sort_test)
  if(TORRENT_RUST_EDITION)
    add_executable(torrent_re_probe rust-edition/sort-probe.cpp)
    target_include_directories(torrent_re_probe PRIVATE src)
    target_compile_definitions(torrent_re_probe PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(torrent_re_probe PRIVATE "${RUST_LIB}" ntdll userenv ws2_32 dbghelp synchronization)
    add_dependencies(torrent_re_probe torrent_re_sort)
    add_test(NAME rust_sort_parity COMMAND torrent_re_probe bench asc)
  endif()
endif()

# The setup shell draws the reference design; NSIS remains the installation backend.
if(EXISTS "${CMAKE_SOURCE_DIR}/dist/Torrent-Payload.exe")
  file(TO_CMAKE_PATH "${CMAKE_SOURCE_DIR}" TORRENT_SOURCE)
  configure_file(installer/setup.rc.in ${CMAKE_BINARY_DIR}/setup.rc @ONLY)
  add_executable(TorrentSetup WIN32 src/setup.cpp ${CMAKE_BINARY_DIR}/setup.rc)
  set_target_properties(TorrentSetup PROPERTIES OUTPUT_NAME "Torrent-${PROJECT_VERSION}-x64")
  target_compile_definitions(TorrentSetup PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(TorrentSetup PRIVATE d2d1 dwrite dwmapi windowscodecs ole32 shell32 uuid advapi32)
  if(MSVC)
    target_compile_options(TorrentSetup PRIVATE /W4 /utf-8 /permissive-)
    target_link_options(TorrentSetup PRIVATE /MANIFEST:NO)
  endif()
endif()
