cmake_minimum_required(VERSION 3.24)
project(TorrentInstallerVisualReview LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(TORRENT_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
configure_file(preview.rc.in ${CMAKE_CURRENT_BINARY_DIR}/preview.rc @ONLY)
add_executable(TorrentInstallerVisualReview WIN32 ../../src/setup.cpp ${CMAKE_CURRENT_BINARY_DIR}/preview.rc)
set_target_properties(TorrentInstallerVisualReview PROPERTIES OUTPUT_NAME "Torrent-Installer-Visual-Review")
target_compile_definitions(TorrentInstallerVisualReview PRIVATE TORRENT_INSTALLER_VISUAL_ONLY UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(TorrentInstallerVisualReview PRIVATE /W4 /utf-8 /permissive-)
target_link_options(TorrentInstallerVisualReview PRIVATE /MANIFEST:NO)
target_link_libraries(TorrentInstallerVisualReview PRIVATE d2d1 dwrite dwmapi windowscodecs ole32 shell32 uuid advapi32 comctl32)
